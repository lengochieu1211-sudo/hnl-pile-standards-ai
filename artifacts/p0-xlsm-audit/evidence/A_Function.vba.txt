Public Function NoiSuyMeyerhof(CaoDo, Range As Range, MuiCoc, DKCoc)
    Dim i As Double
    Dim A1, a3, b1, b3 As Double
    For i = 1 To UBound(Range.Value) * 2 - 3 Step 2
        If Range(i) <= CaoDo And Range(i + 2) >= CaoDo Then
            A1 = Range(i)
            a3 = Range(i + 2)
            b1 = Range(i + 1)
            b3 = Range(i + 3)
            NoiSuyMeyerhof = b1 + (b3 - b1) * (CaoDo - A1) / (a3 - A1)
            If CaoDo = MuiCoc Then
                NoiSuyMeyerhof = (NoiSuy2(CaoDo + 4 * DKCoc, Range) + NoiSuy2(CaoDo - DKCoc, Range)) / 2
            End If
            Exit Function
        End If
    Next
    NoiSuyMeyerhof = 0
End Function

Public Function SoLuongCoc(FZ, HSAT, SCTC, SCT1Coc, SCT2_5Coc, SCT6_10Coc, SCT11_20Coc, SCT21Coc)
    Dim i As Double
    Dim SCT As Double
    For i = 1 To 100
        If i = 1 Then
            SCT = SCTC
        ElseIf i >= 2 And i <= 5 Then
            SCT = 1.15 * SCTC * SCT1Coc / SCT2_5Coc
        ElseIf i >= 6 And i <= 10 Then
            SCT = 1.15 * SCTC * SCT1Coc / SCT6_10Coc
        ElseIf i >= 11 And i <= 20 Then
            SCT = 1.15 * SCTC * SCT1Coc / SCT11_20Coc
        ElseIf i >= 21 Then
            SCT = 1.15 * SCTC * SCT1Coc / SCT21Coc
        End If
        SCT = SCT * i
        If SCT / Math.Abs(FZ) >= HSAT Then
            SoLuongCoc = i
            Exit Function
        End If
    Next
End Function

Public Function HSAT(SLC, SCTC, SCT1Coc, SCT2_5Coc, SCT6_10Coc, SCT11_20Coc, SCT21Coc)
    Dim SCT As Double
    If SLC = 1 Then
        SCT = SCTC
    ElseIf SLC >= 2 And SLC <= 5 Then
        SCT = 1.15 * SCTC * SCT1Coc / SCT2_5Coc
    ElseIf SLC >= 6 And SLC <= 10 Then
        SCT = 1.15 * SCTC * SCT1Coc / SCT6_10Coc
    ElseIf SLC >= 11 And SLC <= 20 Then
        SCT = 1.15 * SCTC * SCT1Coc / SCT11_20Coc
    ElseIf SLC >= 21 Then
        SCT = 1.15 * SCTC * SCT1Coc / SCT21Coc
    End If
    SCT = SCT * SLC
    HSAT = Math.Round(SCT, 0)
End Function

Public Sub DoTimSoLuongCocGanNhau()
    Dim ws As Worksheet
    Set ws = Sheets("TM SCT Coc")
    
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.count, "H").End(xlUp).Row
    
    Dim kccoc As Double
    kccoc = ws.Range("P1").Value
    If kccoc = 0 Then kccoc = 3.1
    
    Dim i As Integer
    i = 19
    Do
        i = i + 1
        Dim TenCoc As Collection
        Set TenCoc = New Collection
    
        ws.Range("A" & i).Value = CountPointsWithinDistance(ws.Range("N" & i).Value, _
                                        ws.Range("O" & i).Value, ws.Range("N20:N" & lastRow), _
                                        ws.Range("O20:O" & lastRow), kccoc * ws.Range("H" & i).Value, TenCoc, ws.Range("C20:C" & lastRow))
        ws.Range("A" & i).Value = TenCoc.count
    Loop Until ws.Range("C" & i + 1).Value = ""
    
End Sub

Function CountPointsWithinDistance(X As Double, Y As Double, RangeX As Range, RangeY As Range, A As Double, ByRef TenCoc As Collection, RangeTenCoc As Range) As Integer
    Dim i As Long
    Dim count As Integer
    Dim deltaX As Double, deltaY As Double, distance As Double
    Dim TC As String
    Dim alreadyExists As Boolean
    
    count = 0
    
    ' Ki?m tra s? lu?ng di?m trong RangeX và RangeY ph?i gi?ng nhau
    If RangeX.Cells.count <> RangeY.Cells.count Then
        CountPointsWithinDistance = -1 ' Tr? v? -1 n?u s? lu?ng không gi?ng nhau
        Exit Function
    End If
    
    ' Duy?t qua t?ng c?p t?a d? trong RangeX và RangeY
    For i = 1 To RangeX.Cells.count
        deltaX = RangeX.Cells(i, 1).Value - X
        deltaY = RangeY.Cells(i, 1).Value - Y
        distance = Sqr(deltaX * deltaX + deltaY * deltaY)
        
        TC = RangeTenCoc.Cells(i, 1).Value
        
        ' Ki?m tra n?u kho?ng cách <= A và TC chua có trong TenCoc
        If distance <= A Then
            alreadyExists = False
            
            ' Ki?m tra xem TC dã có trong TenCoc hay chua
            Dim item As Variant
            For Each item In TenCoc
                If item = TC Then
                    alreadyExists = True
                    Exit For
                End If
            Next item
            ' N?u TC chua có trong TenCoc, thì thêm vào
            If Not alreadyExists Then
                TenCoc.Add TC
                count = count + 1
                count = count + CountPointsWithinDistance(RangeX.Cells(i, 1).Value, RangeY.Cells(i, 1).Value, RangeX, RangeY, A, TenCoc, RangeTenCoc)
            End If
        End If
    Next i
    CountPointsWithinDistance = count
End Function

Public Sub EnsureDceAddinLoaded()
    Dim is64Bit As Boolean
    Dim addinPath As String
    Dim fso As Object
    Dim addin As addin
    Dim addinLoaded As Boolean
    Dim addinFound As Boolean
    Dim addinFileName As String
    
    '--- Bu?c 1: Ki?m tra Excel 32bit hay 64bit ---
    #If Win64 Then
        is64Bit = True
    #Else
        is64Bit = False
    #End If
    
    '--- Ch?n du?ng d?n Add-in ---
    If is64Bit Then
        addinPath = "C:\Dce Pro\V.2020\DCE Excel-AddIn64.xll"
    Else
        addinPath = "C:\Dce Pro\V.2020\DCE Excel-AddIn.xll"
    End If
    
    '--- Ki?m tra file t?n t?i ---
    Set fso = CreateObject("Scripting.FileSystemObject")
    If Not fso.FileExists(addinPath) Then
        MsgBox "Không tìm th?y file add-in: " & vbCrLf & addinPath, vbCritical
        Exit Sub
    End If
    
    '--- L?y tên file ---
    addinFileName = fso.GetFileName(addinPath)
    
    '--- Bu?c 2: Ki?m tra add-in dã du?c load chua ---
    addinFound = False
    addinLoaded = False
    
    For Each addin In Application.AddIns2
        If LCase(fso.GetFileName(addin.FullName)) = LCase(addinFileName) Then
            addinFound = True
            If addin.Installed Then
                addinLoaded = True
            Else
                addin.Installed = True ' Load add-in
                addinLoaded = True
            End If
            Exit For
        End If
    Next addin
    
    '--- N?u chua có thì add ---
    If Not addinFound Then
        Set addin = Application.AddIns2.Add(Filename:=addinPath, CopyFile:=False)
        addin.Installed = True
        addinLoaded = True
    End If
    
    '--- Thông báo ---
    If addinLoaded Then
        Debug.Print "Add-in dã du?c load: " & addinPath
    Else
        MsgBox "Không th? load add-in: " & addinPath, vbCritical
    End If
End Sub
